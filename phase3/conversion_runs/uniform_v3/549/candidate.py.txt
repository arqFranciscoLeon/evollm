def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        self._defect_until_cooperate = False
        return axl.Action.C

    if opponent.history[-1] == axl.Action.D:
        self._defect_until_cooperate = True
        return axl.Action.D

    if self._defect_until_cooperate:
        self._defect_until_cooperate = False
        return axl.Action.C

    return axl.Action.C
