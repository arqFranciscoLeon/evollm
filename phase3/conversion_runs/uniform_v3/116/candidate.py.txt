def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.D

    if opponent.history and opponent.history[-1] == axl.Action.C:
        return axl.Action.D

    if len(opponent.history) >= 5 and opponent.history[-5:] == [axl.Action.D] * 5:
        return axl.Action.D

    if len(self.history) >= 2 and len(opponent.history) >= 2:
        if self.history[-2:] == [axl.Action.D] * 2 and opponent.history[-2:] == [axl.Action.D] * 2:
            return axl.Action.C

    return axl.Action.D
