def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        self._opponent_defect_forever = False
        return axl.Action.C

    if self._opponent_defect_forever:
        return axl.Action.D

    if opponent.history.defections / len(opponent.history) > 0.55:
        self._opponent_defect_forever = True
        return axl.Action.D

    if opponent.history[-1] == axl.Action.D:
        return axl.Action.D

    return axl.Action.C
