def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if len(self.history) + 1 in (998, 999, 1000):
        return axl.Action.D

    if len(opponent.history) >= 1 and opponent.history[-1] == axl.Action.D:
        return axl.Action.D

    if len(opponent.history) >= 2 and opponent.history[-2] == axl.Action.D:
        return axl.Action.D

    if len(opponent.history) >= 10 and opponent.history[-10:] == [axl.Action.C] * 10:
        return axl.Action.D

    return axl.Action.C
