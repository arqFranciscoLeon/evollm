def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.D

    if opponent.history and opponent.history[-1] == axl.Action.D:
        return axl.Action.D

    if 1000 - len(self.history) <= 5:
        return axl.Action.D

    return axl.Action.C
