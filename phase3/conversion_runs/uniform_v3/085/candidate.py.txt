def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.D

    if opponent.history[-1] == axl.Action.C:
        return axl.Action.D

    if opponent.history[-1] == axl.Action.D and self.history[-5:].count(axl.Action.D) == 5:
        return axl.Action.C

    return axl.Action.D
