def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.D

    if len(self.history) == 1:
        return axl.Action.C

    if opponent.history[0] == axl.Action.C and opponent.history[1] == axl.Action.C:
        return axl.Action.D

    return opponent.history[-1]
