def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.D

    current_round = len(self.history) + 1

    if opponent.history and opponent.history[-1] == axl.Action.D:
        return axl.Action.D

    if current_round >= 990:
        return axl.Action.D

    if self.score - opponent.score >= 20:
        return axl.Action.D

    return axl.Action.C
