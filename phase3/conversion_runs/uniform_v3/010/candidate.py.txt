def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.C

    if opponent.history.defections > len(opponent.history) / 2:
        return axl.Action.D

    if opponent.history and opponent.history[-1] == axl.Action.D:
        return axl.Action.D

    return axl.Action.C
