def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.C

    last_my_payoff, _ = self.total_scores(self.history[-1:], opponent.history[-1:])
    if last_my_payoff in (3, 5):
        return self.history[-1]
    if last_my_payoff in (0, 1):
        return axl.Action.C if self.history[-1] == axl.Action.D else axl.Action.D

    return axl.Action.C
