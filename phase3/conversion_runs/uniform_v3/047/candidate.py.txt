def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.D

    if opponent.history and opponent.history[-1] == axl.Action.C:
        return axl.Action.D

    if len(opponent.history) >= 3 and opponent.history[-3:] == [axl.Action.D, axl.Action.D, axl.Action.D]:
        return axl.Action.C

    return axl.Action.D
