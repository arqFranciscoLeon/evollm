def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        return axl.Action.C

    if len(opponent.history) >= 10 and opponent.history[-10:].count(axl.Action.D) >= 8:
        return axl.Action.D

    if opponent.history[-1] == axl.Action.C:
        return axl.Action.C

    return axl.Action.D
