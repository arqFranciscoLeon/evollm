def strategy(self, opponent: axl.player.Player) -> axl.Action:
    if self.first_round():
        self._opponent_always_defect = False
        return axl.Action.C

    if self._opponent_always_defect:
        return axl.Action.D

    if len(opponent.history) > 0 and opponent.history.defections / len(opponent.history) > 0.55:
        self._opponent_always_defect = True
        return axl.Action.D

    if opponent.history[-1] == axl.Action.D:
        return axl.Action.D

    return axl.Action.C
